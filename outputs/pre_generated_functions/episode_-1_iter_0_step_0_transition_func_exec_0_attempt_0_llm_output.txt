def transition_func(state, action):
    """
    Args:
        state (MinigridState): the state of the environment 
        action (int): action to be taken in state `state`
    Returns:
        new_state (MinigridState): the new state of the environment
    """

    def rotate_left(agent_dir):
        return (agent_dir - 1) % 4

    def rotate_right(agent_dir):
        return (agent_dir + 1) % 4

    def move_forward(agent_pos, agent_dir):
        new_pos = (
            agent_pos[0] + DIR_TO_VEC[agent_dir][0],
            agent_pos[1] + DIR_TO_VEC[agent_dir][1]
        )
        if 0 <= new_pos[0] < state.width and 0 <= new_pos[1] < state.height and state.grid[new_pos] != ObjectTypes.wall:
            return new_pos
        return agent_pos

    def pickup_object(agent_pos, carrying):
        grid_value = state.grid[agent_pos]
        if grid_value in [ObjectTypes.key, ObjectTypes.ball, ObjectTypes.box] and carrying is None:
            return grid_value
        return carrying

    def drop_object(agent_pos, carrying):
        if carrying is not None:
            state.grid[agent_pos] = carrying
            return None
        return carrying

    def toggle_object(agent_pos, agent_dir):
        front_pos = (
            agent_pos[0] + DIR_TO_VEC[agent_dir][0],
            agent_pos[1] + DIR_TO_VEC[agent_dir][1]
        )
        if 0 <= front_pos[0] < state.width and 0 <= front_pos[1] < state.height:
            front_value = state.grid[front_pos]
            if front_value in [ObjectTypes.open_door, ObjectTypes.closed_door, ObjectTypes.locked_door]:
                return not (front_value == ObjectTypes.open_door)
        return False

    new_agent_pos = state.agent_pos
    new_agent_dir = state.agent_dir
    new_carrying = state.carrying

    if action == Actions.left:
        new_agent_dir = rotate_left(new_agent_dir)
    elif action == Actions.right:
        new_agent_dir = rotate_right(new_agent_dir)
    elif action == Actions.forward:
        new_agent_pos = move_forward(new_agent_pos, new_agent_dir)
    elif action == Actions.pickup:
        new_carrying = pickup_object(new_agent_pos, new_carrying)
    elif action == Actions.drop:
        new_carrying = drop_object(new_agent_pos, new_carrying)
    elif action == Actions.toggle:
        if toggle_object(new_agent_pos, new_agent_dir):
            front_pos = (
                new_agent_pos[0] + DIR_TO_VEC[new_agent_dir][0],
                new_agent_pos[1] + DIR_TO_VEC[new_agent_dir][1]
            )
            state.grid[front_pos] = ObjectTypes.open_door if state.grid[front_pos] == ObjectTypes.closed_door else ObjectTypes.closed_door

    return MinigridState(
        grid=state.grid.copy(),
        agent_pos=new_agent_pos,
        agent_dir=new_agent_dir,
        carrying=new_carrying
    )
